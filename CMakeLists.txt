cmake_minimum_required(VERSION 3.28)
project(SourceCubemap VERSION 0.1.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "SourceCubemap currently requires Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(CLI11 CONFIG REQUIRED)
find_package(spdlog CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(xxHash CONFIG REQUIRED)
find_package(glm CONFIG REQUIRED)
find_package(volk CONFIG REQUIRED)
find_package(VulkanMemoryAllocator CONFIG REQUIRED)
include(FetchContent)
set(BZIP2_TAG f5dfc96063fef2d9c5446d0733867d899ef9b942 CACHE STRING "Pinned minizip bzip2 dependency" FORCE)
set(LIBLZMA_TAG 299eb5c3f597f107fd6019bf156885ac8bfaa5ea CACHE STRING "Pinned minizip LZMA dependency" FORCE)
set(PPMD_TAG f9d78aff31a5f2521ae7ddbdc97c4a8855808959 CACHE STRING "Pinned minizip PPMd dependency" FORCE)
set(ZLIB_TAG 12731092979c6d07f42da27da673a9f6c7b13586 CACHE STRING "Pinned minizip zlib dependency" FORCE)
set(ZSTD_TAG f8745da6ff1ad1e7bab384bd1f9d742439278e99 CACHE STRING "Pinned minizip zstd dependency" FORCE)
set(FETCHCONTENT_UPDATES_DISCONNECTED ON CACHE BOOL "Use pinned dependency revisions without update polling")
set(SOURCEPP_LIBS_START_ENABLED OFF CACHE BOOL "" FORCE)
foreach(lib BSPPP KVPP MDLPP VPKPP VTFPP STEAMPP)
    set(SOURCEPP_USE_${lib} ON CACHE BOOL "" FORCE)
endforeach()
set(SOURCEPP_BUILD_WIN7_COMPAT OFF CACHE BOOL "" FORCE)
set(SOURCEPP_LINK_STATIC_MSVC_RUNTIME ON CACHE BOOL "" FORCE)
set(SOURCEPP_VPKPP_SUPPORT_VPK_V54 OFF CACHE BOOL "" FORCE)
foreach(format EXR JXL QOI WEBP)
    set(SOURCEPP_VTFPP_SUPPORT_${format} OFF CACHE BOOL "" FORCE)
endforeach()
FetchContent_Declare(sourcepp GIT_REPOSITORY https://github.com/craftablescience/sourcepp.git GIT_TAG 1540277f99bacee288f6786a66ee182aab6c146b)
FetchContent_MakeAvailable(sourcepp)
add_library(scm_core src/common.cpp src/bsp.cpp src/assets.cpp)
target_include_directories(scm_core PUBLIC include)
target_compile_definitions(scm_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN GLM_FORCE_DEPTH_ZERO_TO_ONE)
target_link_libraries(scm_core PUBLIC sourcepp::bsppp sourcepp::vpkpp sourcepp::vtfpp sourcepp::kvpp sourcepp::mdlpp sourcepp::steampp glm::glm nlohmann_json::nlohmann_json spdlog::spdlog xxHash::xxhash)
add_executable(sourcecubemap src/main.cpp)
add_library(scm_scene src/scene.cpp src/cache.cpp)
target_link_libraries(scm_scene PUBLIC scm_core)
add_library(scm_capture src/capture.cpp)
target_link_libraries(scm_capture PUBLIC scm_core)
add_library(scm_renderer src/renderer.cpp)
target_link_libraries(scm_renderer PUBLIC scm_scene scm_capture volk::volk GPUOpen::VulkanMemoryAllocator)
set(SCM_DXC_EXECUTABLE "" CACHE FILEPATH "SPIR-V-capable DXC override")
if(SCM_DXC_EXECUTABLE)
    set(DXC_EXECUTABLE "${SCM_DXC_EXECUTABLE}")
else()
    FetchContent_Declare(scm_dxc
        URL https://github.com/microsoft/DirectXShaderCompiler/releases/download/v1.9.2609/dxc_2026_09_29.zip
        URL_HASH SHA256=ad31b1fc8443175d204f77a611fdb3ef2ec42759bdc2f1167368de24a4a7e7f1)
    FetchContent_MakeAvailable(scm_dxc)
    set(DXC_EXECUTABLE "${scm_dxc_SOURCE_DIR}/bin/x64/dxc.exe")
endif()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/shaders")
foreach(stage vert frag)
    if(stage STREQUAL "vert")
        set(entry VSMain)
        set(profile vs_6_0)
    else()
        set(entry PSMain)
        set(profile ps_6_0)
    endif()
    add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/shaders/scene.${stage}.spv"
        COMMAND "${DXC_EXECUTABLE}" -spirv -fspv-target-env=vulkan1.3 -E ${entry} -T ${profile} -O3
            -Fo "${CMAKE_CURRENT_BINARY_DIR}/shaders/scene.${stage}.spv" "${CMAKE_CURRENT_SOURCE_DIR}/shaders/scene.hlsl"
        DEPENDS shaders/scene.hlsl VERBATIM)
    list(APPEND compiled_shaders "${CMAKE_CURRENT_BINARY_DIR}/shaders/scene.${stage}.spv")
endforeach()
add_custom_target(scm_shaders DEPENDS ${compiled_shaders})
add_dependencies(sourcecubemap scm_shaders)
target_link_libraries(sourcecubemap PRIVATE scm_renderer CLI11::CLI11)
if(MSVC)
    target_compile_options(scm_core PRIVATE /W4 /permissive- /EHsc /utf-8)
    target_compile_options(sourcecubemap PRIVATE /W4 /permissive- /EHsc /utf-8)
endif()
install(TARGETS sourcecubemap RUNTIME DESTINATION bin)
install(FILES ${compiled_shaders} DESTINATION bin/shaders)
